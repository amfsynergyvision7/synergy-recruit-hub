-- BUILD160 REVIEW ONLY / DO NOT APPLY / NOT A PRODUCTION MIGRATION.
-- Owner approval must precede any change to legitimate audit event writers.
-- Preserve the independently reviewed administrator SELECT policy. Revoke
-- anonymous reads entirely and make service_role append-only (INSERT, not
-- UPDATE/DELETE/TRUNCATE). A database owner/break-glass operator is separate.
-- Build163 also revokes REFERENCES, TRIGGER and MAINTAIN for each app role.
REVOKE SELECT,INSERT,UPDATE,DELETE,TRUNCATE ON TABLE public.audit_logs
 FROM PUBLIC,anon;
REVOKE INSERT,UPDATE,DELETE,TRUNCATE ON TABLE public.audit_logs
 FROM authenticated;
REVOKE UPDATE,DELETE,TRUNCATE ON TABLE public.audit_logs
 FROM service_role;
-- Build163: PostgreSQL 17 treats REFERENCES, TRIGGER and MAINTAIN as
-- independent table privileges. RLS cannot protect these grant surfaces.
-- Remove them for every non-owner app role, including the trusted writer.
REVOKE REFERENCES,TRIGGER,MAINTAIN ON TABLE public.audit_logs
 FROM PUBLIC,anon,authenticated,service_role;
DROP POLICY IF EXISTS "Allow all for authenticated" ON public.audit_logs;
DROP POLICY IF EXISTS "insert audit" ON public.audit_logs;
